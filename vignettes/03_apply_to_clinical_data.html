Overview

This vignette demonstrates how to apply the pmsimstats2025 analysis methodology to actual clinical trial data. The workflow covers:

  1. Preparing clinical trial data for analysis
  2. Defining the trial design structure
  3. Fitting mixed-effects models with biomarker moderation
  4. Generating personalized treatment response predictions
  5. Visualizing individual patient outcomes

Create Design Matrix

# ============================================================================
# CREATE DESIGN MATRIX
# ============================================================================

create_trial_design_matrix <- function(measurement_weeks, n_paths = 4) {
  # Create full factorial of weeks and paths
  design <- expand_grid(
    path = 1:n_paths,
    week = measurement_weeks
  ) |>
    mutate(
      # Hybrid design treatment assignment rules
      treatment = case_when(
        week %in% c(4, 8) ~ 1,
        week == 9 ~ 1,
        week == 10 & path %in% c(1, 2) ~ 1,
        week == 10 & path %in% c(3, 4) ~ 0,
        week %in% c(11, 12) ~ 0,
        week == 16 & path %in% c(1, 3) ~ 1,
        week == 16 & path %in% c(2, 4) ~ 0,
        week == 20 & path %in% c(1, 3) ~ 0,
        week == 20 & path %in% c(2, 4) ~ 1,
        TRUE ~ NA_real_
      ),
      # Expectancy (blinding) status
      expectancy = if_else(week %in% c(4, 8), 1.0, 0.5),
      # Timepoint index
      timepoint = match(week, measurement_weeks)
    )

  return(design)
}

design_matrix <- create_trial_design_matrix(measurement_weeks)

Part 2: Prepare Clinical Data

Required Data Format

Clinical trial data must be in long format with the following columns:

Column Description
participant_id Unique participant identifier
biomarker Baseline biomarker value
baseline Baseline symptom severity
path Randomization path (1-4)
week Measurement week
response Symptom score at measurement 
# ============================================================================
# LOAD CLINICAL DATA
# ============================================================================

# Example: Load clinical trial data
# In practice, replace with your actual data file
# clinical_data <- read_csv("path/to/clinical_data.csv")

# For demonstration, generate synthetic clinical data
set.seed(42)
n_participants <- 50

# Generate participant-level data
participant_info <- tibble(
  participant_id = 1:n_participants,
  biomarker = rnorm(n_participants, mean = 5, sd = 2),
  baseline = rnorm(n_participants, mean = 10, sd = 2),
  path = sample(1:4, n_participants, replace = TRUE)
)

# Merge with design matrix
clinical_data <- participant_info |>
  left_join(design_matrix, by = "path", relationship = "many-to-many") |>
  mutate(
    # Simulate responses based on model
    bm_centered = (biomarker - mean(biomarker)) / sd(biomarker),
    BR_rate = 0.5 * (1 + 0.3 * bm_centered),
    weeks_on_drug = cumsum(treatment),
    response = baseline +
      weeks_on_drug * BR_rate * treatment +
      0.2 * week * expectancy +
      0.1 * week +
      rnorm(n(), 0, 1.5)
  ) |>
  select(participant_id, biomarker, baseline, path, week, response)

Part 3: Fit Mixed-Effects Model

Model Specification

The analysis uses a linear mixed-effects model with:

  • Fixed effects: treatment, biomarker, treatment × biomarker interaction, week
  • Random effects: participant-specific intercept
  • Correlation structure: CAR(1) for repeated measures
# ============================================================================
# FIT MIXED-EFFECTS MODEL
# ============================================================================

# Prepare data for modeling
model_data <- clinical_data |>
  left_join(design_matrix, by = c("path", "week")) |>
  mutate(
    bm_centered = biomarker - mean(biomarker),
    participant_id = factor(participant_id)
  )

# Fit LME model with biomarker × treatment interaction
model_fit <- lme(
  response ~ treatment * bm_centered + week,
  random = ~ 1 | participant_id,
  correlation = corCAR1(form = ~ week | participant_id),
  data = model_data,
  control = lmeControl(opt = "optim")
)

biomarker_range <- seq(
  min(clinical_data$biomarker),
  max(clinical_data$biomarker),
  length.out = 50
)

# Predictions for treatment ON vs OFF
prediction_grid <- expand_grid(
  bm_centered = biomarker_range - mean(clinical_data$biomarker),
  treatment = c(0, 1),
  week = max(measurement_weeks)
) |>
  mutate(
    biomarker = bm_centered + mean(clinical_data$biomarker)
  )

# Get population-level predictions (fixed effects only)
prediction_grid$predicted <- predict(
  model_fit,
  newdata = prediction_grid,
  level = 0  # Population level (no random effects)
)

# Calculate treatment effect as function of biomarker
treatment_effect_by_bm <- prediction_grid |>
  select(biomarker, treatment, predicted) |>
  pivot_wider(
    names_from = treatment,
    values_from = predicted,
    names_prefix = "trt_"
  ) |>
  mutate(
    treatment_effect = trt_1 - trt_0
  )

Personalized Response Prediction

# ============================================================================
# PERSONALIZED PREDICTION FOR NEW PATIENT
# ============================================================================

predict_patient_response <- function(model, biomarker_value, baseline_value,
                                     treatment_weeks = c(4, 8, 9, 10)) {
  # Prepare prediction data
  mean_biomarker <- mean(clinical_data$biomarker)

  pred_data <- tibble(
    week = measurement_weeks,
    bm_centered = biomarker_value - mean_biomarker,
    treatment = if_else(week %in% treatment_weeks, 1, 0)
  )

  # Generate predictions
  pred_data$predicted <- predict(model, newdata = pred_data, level = 0)

  # Adjust for baseline
  baseline_adjustment <- baseline_value - pred_data$predicted[1]
  pred_data$predicted <- pred_data$predicted + baseline_adjustment

  return(pred_data)
}

# Example: Predict for a new patient
new_patient <- predict_patient_response(
  model = model_fit,
  biomarker_value = 7.5,
  baseline_value = 12.0
)
