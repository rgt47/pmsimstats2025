Overview

This vignette demonstrates how to analyze and visualize simulation results from pmsimstats2025. The workflow covers:

  1. Loading and exploring simulation results
  2. Computing summary statistics (power, effect sizes)
  3. Creating power heatmaps
  4. Comparing designs with and without carryover modeling
  5. Generating publication-quality figures

Part 3: Create Power Heatmaps

Single Design Heatmap

# ============================================================================
# SINGLE DESIGN HEATMAP
# ============================================================================

plot_power_heatmap_single <- function(data, design_name) {
  design_labels <- c(
    "ol_bdc" = "OL+BDC\n(Design 2)",
    "hybrid" = "Hybrid/N-of-1\n(Design 4)"
  )

  plot_data <- data %>%
    filter(design == design_name) %>%
    mutate(
      carryover = factor(carryover),
      biomarker_moderation = factor(
        biomarker_moderation,
        levels = rev(sort(unique(biomarker_moderation)))
      )
    )

  p <- ggplot(
    plot_data,
    aes(x = carryover, y = biomarker_moderation, fill = power)
  ) +
    geom_tile(color = "white", linewidth = 0.5) +
    geom_text(
      aes(label = sprintf("%.0f%%", power * 100)),
      color = "black",
      size = 4,
      fontface = "bold"
    ) +
    scale_fill_gradient2(
      name = "Power",
      low = "#d73027",
      mid = "#fee08b",
      high = "#1a9850",
      midpoint = 0.5,
      limits = c(0, 1),
      labels = scales::percent
    ) +
    scale_x_discrete(expand = c(0, 0)) +
    scale_y_discrete(expand = c(0, 0)) +
    labs(
      title = design_labels[design_name],
      x = "Carryover (proportion retained)",
      y = "Biomarker Moderation\n(effect size)"
    ) +
    theme_minimal(base_size = 12) +
    theme(
      panel.grid = element_blank(),
      axis.ticks = element_blank(),
      plot.title = element_text(hjust = 0.5, face = "bold"),
      legend.position = "right"
    )

  return(p)
}

# Create heatmap for hybrid design
p_hybrid <- plot_power_heatmap_single(summary_results, "hybrid")
print(p_hybrid)

Multi-Design Comparison

# ============================================================================
# MULTI-DESIGN HEATMAP
# ============================================================================

plot_power_heatmap_multi <- function(data) {
  design_labels <- c(
    "ol_bdc" = "OL+BDC (Design 2)",
    "hybrid" = "Hybrid/N-of-1 (Design 4)"
  )

  plot_data <- data %>%
    mutate(
      design = factor(
        design_labels[as.character(design)],
        levels = design_labels
      ),
      carryover = factor(carryover),
      biomarker_moderation = factor(
        biomarker_moderation,
        levels = rev(sort(unique(biomarker_moderation)))
      )
    )

  p <- ggplot(
    plot_data,
    aes(x = carryover, y = biomarker_moderation, fill = power)
  ) +
    geom_tile(color = "white", linewidth = 0.5) +
    geom_text(
      aes(label = sprintf("%.0f%%", power * 100)),
      color = "black",
      size = 3.5,
      fontface = "bold"
    ) +
    scale_fill_gradient2(
      name = "Power",
      low = "#d73027",
      mid = "#fee08b",
      high = "#1a9850",
      midpoint = 0.5,
      limits = c(0, 1),
      labels = scales::percent
    ) +
    scale_x_discrete(expand = c(0, 0)) +
    scale_y_discrete(expand = c(0, 0)) +
    facet_wrap(~design, ncol = 2) +
    labs(
      title = "Statistical Power - Clustered Designs",
      subtitle = "N=70 participants, 100 iterations per condition",
      x = "Carryover (proportion retained)",
      y = "Biomarker Moderation (effect size)"
    ) +
    theme_minimal(base_size = 12) +
    theme(
      panel.grid = element_blank(),
      axis.ticks = element_blank(),
      plot.title = element_text(hjust = 0.5, face = "bold"),
      plot.subtitle = element_text(hjust = 0.5),
      legend.position = "right",
      strip.text = element_text(face = "bold", size = 12),
      strip.background = element_rect(fill = "grey90", color = NA)
    )

  return(p)
}

# Create multi-panel heatmap
p_multi <- plot_power_heatmap_multi(summary_results)
print(p_multi)

Part 5: Carryover Impact Analysis

# ============================================================================
# CARRYOVER IMPACT ANALYSIS
# ============================================================================

# Calculate power change with carryover
carryover_impact <- summary_results %>%
  filter(biomarker_moderation > 0) %>%
  select(design, biomarker_moderation, carryover, power) %>%
  pivot_wider(
    names_from = carryover,
    values_from = power,
    names_prefix = "co_"
  ) %>%
  mutate(
    power_loss_50 = co_0 - co_0.5,
    power_loss_100 = co_0 - co_1,
    pct_loss_50 = (co_0 - co_0.5) / co_0 * 100,
    pct_loss_100 = (co_0 - co_1) / co_0 * 100
  )
